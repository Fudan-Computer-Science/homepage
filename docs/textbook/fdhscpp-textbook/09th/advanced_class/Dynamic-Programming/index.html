fibonacci sequence​

遞迴​

最直覺的解法

long long fib(int x) {
	if (x < 2) return x;
	return fib(x - 1) + fib(x - 2);
}

時間複雜度O(ϕn)O(\phi^n)O(ϕn)

轉移式​

dpi={ii<2dpi−1+dpi−2otherwisedp_i=\begin{cases}i&i<2\\dp_{i-1}+dp_{i-2}&otherwise\end{cases}dpi​={idpi−1​+dpi−2​​i<2otherwise​

DP解​

將答案存起來避免重複計算

long long ans[100];
long long fib(int x) {
	if (x < 2) return x;
	return ans[x] ? ans[x] : (ans[x] = fib(x - 1) + fib(x - 2));
}

時間複雜度O(n)O(n)O(n), 空間複雜度O(n)O(n)O(n)

滾動DP​

long long fib(int x) {
	vector<int> v{0, 1};
	for (int i = 2; i <= x; i++)
		v[i & 1] += v[~i & 1];
	return v[x & 1];
}

時間複雜度O(n)O(n)O(n), 空間複雜度O(1)O(1)O(1)

Longest Common Subsequence(LCS)​

轉移式​

dpi,j={1+dpi−1,j−1ai=bjmax(dpi−1,j,dpi,j−1)otherwisedp_{i,j}=\begin{cases}1+dp_{i-1,j-1}&a_i=b_j\\max(dp_{i-1,j},dp_{i,j-1})&otherwise\end{cases}dpi,j​={1+dpi−1,j−1​max(dpi−1,j​,dpi,j−1​)​ai​=bj​otherwise​

DP解​

vector<vector<int>> dp(a.size() + 1, vector<int>(b.size() + 1));
for (size_t i = 0; i < a.size(); i++)
	for (size_t j = 0; j < b.size(); j++) {
		if (a[i] == b[j]) dp[i + 1][j + 1] = 1 + dp[i][j];
		else dp[i + 1][j + 1] = max(dp[i][j + 1], dp[i + 1][j]);
	}
cout << dp[a.size()][b.size()] << endl;

時間複雜度O(nm)O(nm)O(nm), 空間複雜度O(nm)O(nm)O(nm)

滾動DP​

auto& a = max(inputa, inputb, [](auto& x, auto& y) {return x.size() < y.size();});
auto& b = min(inputa, inputb, [](auto& x, auto& y) {return x.size() < y.size();});
vector<vector<int>> dp(2, vector<int>(b.size() + 1));
for (size_t i = 0; i < a.size(); i++)
	for (size_t j = 0; j < b.size(); j++) {
		if (a[i] == b[j]) dp[~i & 1][j + 1] = 1 + dp[i & 1][j];
		else dp[~i & 1][j + 1] = max(dp[i & 1][j + 1], dp[~i & 1][j]);
	}
cout << dp[a.size() & 1][b.size()] << endl;

c001: 10405 - Longest Common Subsequence

路徑總數​

轉移式​

dpi,j={1i=1 or j=1dpi−1,j+dpi,j−1otherwisedp_{i,j}=\begin{cases}1&i=1\ or\ j=1\\dp_{i-1,j}+dp_{i,j-1}&otherwise\end{cases}dpi,j​={1dpi−1,j​+dpi,j−1​​i=1 or j=1otherwise​

DP解​

vector<vector<int>> dp(n, vector<int>(m, 1));
for (int i = 1; i < n; i++)
	for (int j = 1; j < m; j++)
		dp[i][j] = (dp[i - 1][j] + dp[i][j - 1]) % mod;
cout << dp.back().back() << endl;

時間複雜度O(nm)O(nm)O(nm), 空間複雜度O(nm)O(nm)O(nm)

滾動DP​

auto n = max(a, b);
auto m = min(a, b);
vector<vector<int>> dp(2, vector<int>(m, 1));
for (int i = 1; i < n; i++)
	for (int j = 1; j < m; j++)
		dp[i & 1][j] = dp[~i & 1][j] + dp[i & 1][j - 1];
cout << dp[~n & 1].back() << endl;

時間複雜度O(nm)O(nm)O(nm), 空間複雜度O(min(n,m))O(min(n,m))O(min(n,m))

範例code
//沒在OJ上試過，可能有些小bug
#include<iostream>
#include<vector>
#include<functional>
using namespace std;
const string dic = "AUCG";
struct node {
	string val;
	vector<long long> ch, dp;
};
int main() {
	int n, m, a, b, root;
	cin >> n >> m;
	vector<node> v(n);
	for (int i = 0; i != n; i++) {
		cin >> a >> b >> v[a - 1].val;
		if (a == b) root = a - 1;
		else v[b - 1].ch.emplace_back(a - 1);
	}
	long long sum = 0;
	for (int k = 0; k != m; k++) {
		static function<void(int)> dfs = [&](int x) {
			v[x].dp.assign(4, 0);
			for (const auto& i : v[x].ch) dfs(i);
			for (int i = 0; i != 4; i++) {
				if (v[x].val[k] != '@' && v[x].val[k] != dic[i]) {
					v[x].dp[i] = 1e9; continue;
				}
				for (auto& j : v[x].ch) {
					long long minn = 1e9;
					for (int l = 0; l != 4; l++)
						minn = min(minn, v[j].dp[l] + (l != i));
					v[x].dp[i] += minn;
				}
			}
		};
		dfs(root), sum += min({v[root].dp[0], v[root].dp[1], v[root].dp[2], v[root].dp[3]});
	}
	cout << sum << endl;
	return 0;
}
